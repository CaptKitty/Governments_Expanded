GE_calculate_dinastic_power = {
	hidden_effect = {
		export_to_variable = {
			which = adm_value
			value = ADM
		}
		export_to_variable = {
			which = dip_value
			value = DIP
		}
		export_to_variable = {
			which = mil_value
			value = MIL
		}
		change_variable = { which = adm_value which = dip_value }
		change_variable = { which = adm_value which = mil_value }
		multiply_variable = {
			which = adm_value
			value = 1.5
		}
		while = {
			limit = {
				check_variable = {
					which = adm_value
					value = 1
				}
			}
			if = {
				limit = { has_government_mechanic = italian_dinasties_mechanic }
				add_government_power = {
					mechanic_type = italian_dinasties_mechanic
					power_type = italian_dinasties
					value = 1
				}
			}
			else_if = {
				limit = { has_government_mechanic = sicilian_kingdom_mechanic }
				add_government_power = {
					mechanic_type = sicilian_kingdom_mechanic
					power_type = sicilian_kingdom_adm_power
					value = 1
				}
				add_government_power = {
					mechanic_type = sicilian_kingdom_mechanic
					power_type = sicilian_kingdom_dip_power
					value = 1
				}
				add_government_power = {
					mechanic_type = sicilian_kingdom_mechanic
					power_type = sicilian_kingdom_mil_power
					value = 1
				}
			}
			else_if = {
				limit = { has_government_mechanic = dukes_of_brittany_mechanic }
				add_government_power = {
					mechanic_type = dukes_of_brittany_mechanic
					power_type = breton_tradition
					value = 1
				}
			}
			else_if = {
				limit = { has_government_mechanic = imperial_control_mechanic }
				add_government_power = {
					mechanic_type = imperial_control_mechanic
					power_type = imperial_control
					value = 1
				}
			}
			change_variable = { which = adm_value value = -1 }
		}
	}
}

GE_clear_flag = {
	clr_country_flag = GE_integrating_$culture$
}
GE_clear_integration_flag = {
	GE_execute_effect_for_every_culture = {
		effect = GE_clear_flag
	}
}
GE_execute_effect_for_every_culture = {
	$effect$ = { culture = flemish }
	$effect$ = { culture = wallonian }
	$effect$ = { culture = dutch }
	$effect$ = { culture = hessian }
}

GE_cultural_unity_integrated = {
	GE_execute_effect_for_every_culture = {
		effect = GE_cultural_unity_integrated_specific
	}

	hidden_effect = {
		GE_clear_integration_flag = yes
		regenerate_government_mechanics = yes

		set_government_power = {
			mechanic_type = cultural_disunity_mechanic
			power_type = cultural_disunity
			value = 0
		}
	}
}

GE_cultural_unity_integrated_specific = {
	if = {
		limit = {
			has_country_flag = GE_integrating_$culture$
		}
		set_country_flag = GE_integrated_$culture$
		GE_integrated_culture = { culture = $culture$ }
	}
}

GE_integrated_culture = {
	if = {
		limit = {
			accepted_culture = $culture$
		}
		add_country_modifier = {
			name = GE_$culture$_modifier
			duration = -1
			desc = GE_until_demotion
		}
	}
	else = {
		add_country_modifier = {
			name = GE_$culture$_modifier
			duration = -1
			desc = GE_until_demotion
		}
		add_accepted_culture = $culture$
	}
}


GE_on_culture_promoted = {
	GE_execute_effect_for_every_culture = {
		effect = GE_on_specific_culture_promoted
	}
}

GE_on_specific_culture_promoted = {
	if = {
		limit = {
			accepted_culture = $culture$
			has_country_flag = GE_integrated_$culture$
			NOT = { has_country_modifier = GE_$culture$_modifier }
		}
		add_country_modifier = {
			name = GE_$culture$_modifier
			duration = -1
			desc = GE_until_demotion
		}
	}
}

GE_on_culture_demoted = {
	GE_execute_effect_for_every_culture = {
		effect = GE_on_specific_culture_demoted
	}
}

GE_on_specific_culture_demoted = {
	if = {
		limit = {
			OR = {
				NOT = { has_country_flag = GE_integrated_$culture$ }
				NOT = { accepted_culture = $culture$ }
			}
			has_country_modifier = GE_$culture$_modifier
		}
		remove_country_modifier = GE_$culture$_modifier
	}
}

GE_start_integrating = {
	hidden_effect = {
		GE_clear_integration_flag = yes
	}

	custom_tooltip = GE_begin_integrating

	set_country_flag = GE_integrating_$culture$

	hidden_effect = {
		regenerate_government_mechanics = yes
		unfreeze_government_power = {
			mechanic_type = cultural_disunity_mechanic
			power_type = cultural_disunity
		}
	}
}
