country_event = {
	id = GE_Mechanic_Events.10
	title = GE_Mechanic_Events.10.T
	desc = GE_Mechanic_Events.10.D
	picture = CULTURE_eventPicture

	is_triggered_only = yes

	option = {
		name = GE_Mechanic_Events.10.OPT1
		ai_chance = { factor = 1 }

		GE_cultural_unity_integrated = yes

		if = {
			limit = {
				GE_num_of_cultures_accepted = { amount = 4 }
			}
			change_government_reform = ge_communities_regions_reform
		}
	}

	after = {
		freeze_government_power = {
			mechanic_type = cultural_disunity_mechanic
			power_type = cultural_disunity
		}
	}
}
